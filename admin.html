/**
 * Booth Azzuha Headcount Backend — Google Apps Script
 * Sasaran Sheet: https://docs.google.com/spreadsheets/d/1tQ6fhzWnjhJk2aiJhQAaTmA5tVymbfyeDZTxP3QjjF8/edit
 */

const SPREADSHEET_ID = '1tQ6fhzWnjhJk2aiJhQAaTmA5tVymbfyeDZTxP3QjjF8';

function getTargetSpreadsheet() {
  return SpreadsheetApp.openById(SPREADSHEET_ID);
}

function sanitizeSheetName(name) {
  const clean = String(name || 'Booth_Azzuha')
    .replace(/[\[\]\*\/\\\?\:]/g, '')
    .trim();
  return clean.substring(0, 95) || 'Booth_Azzuha';
}

function getOrCreateEventSheet(eventName) {
  const ss = getTargetSpreadsheet();
  const sheetName = sanitizeSheetName(eventName);
  let sheet = ss.getSheetByName(sheetName);
  
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
    sheet.appendRow(['ID', 'Tarikh_Masa', 'Pakej', 'Nama_Jemaah', 'No_Telefon', 'Pax']);
    sheet.getRange('A1:F1').setFontWeight('bold').setBackground('#d1fae5');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function doGet(e) {
  try {
    const ss = getTargetSpreadsheet();
    const props = PropertiesService.getScriptProperties();
    const activeEvent = props.getProperty('ACTIVE_EVENT') || '';
    const sheets = ss.getSheets();
    const records = [];

    sheets.forEach(sheet => {
      const sheetName = sheet.getName();
      const rows = sheet.getDataRange().getValues();
      if (rows.length < 2) return;
      if (String(rows[0][0]) !== 'ID') return;

      for (let i = 1; i < rows.length; i++) {
        if (!rows[i][0]) continue;
        records.push({
          id: String(rows[i][0]),
          timestamp: String(rows[i][1]),
          eventName: sheetName,
          pakej: String(rows[i][2]),
          nama: String(rows[i][3]),
          phone: String(rows[i][4]),
          pax: Number(rows[i][5]) || 1
        });
      }
    });

    return ContentService
      .createTextOutput(JSON.stringify({ success: true, activeEvent: activeEvent, records: records.reverse() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ success: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.tryLock(10000);

  try {
    const data = JSON.parse(e.postData.contents);
    if (data.action === 'set_event') {
      const evName = data.eventName || 'Booth Azzuha';
      PropertiesService.getScriptProperties().setProperty('ACTIVE_EVENT', evName);
      const createdSheet = getOrCreateEventSheet(evName);
      return ContentService
        .createTextOutput(JSON.stringify({ success: true, activeEvent: evName, tab: createdSheet.getName() }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const sheet = getOrCreateEventSheet(data.eventName || 'Booth Azzuha');
    
    const id = data.id || ('AZH-' + Math.floor(1000 + Math.random() * 9000));
    const masa = data.timestamp || Utilities.formatDate(new Date(), 'Asia/Kuala_Lumpur', 'dd/MM/yyyy HH:mm');

    sheet.appendRow([
      id,
      masa,
      data.pakej || 'Umrah',
      data.nama || '',
      "'" + (data.phone || ''),
      Number(data.pax) || 1
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({ success: true, id: id, tab: sheet.getName() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ success: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
