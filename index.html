Sambungan Google Sheet

Tampal URL Web App Google Apps Script anda supaya semua nama jemaah terus masuk ke dalam Google Sheet.

Kod & URL Google Apps Script

Salin kod backend dan tetapkan URL terus dalam kod sistem

const SHEET_NAME = 'Headcount_Log';

function setupSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow(['Timestamp','Date','Time','Entry Type','Pax Count','Visitor Name','Phone','Interest / Package','Staff / Booth PIC','Notes']);
    sheet.getRange(1, 1, 1, 10).setFontWeight('bold').setBackground('#f3f4f6');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function doGet() {
  const sheet = setupSheet();
  const rows = sheet.getDataRange().getValues().slice(1);
  const tz = Session.getScriptTimeZone();
  const todayStr = Utilities.formatDate(new Date(), tz, 'yyyy-MM-dd');
  let totalPax = 0, todayPax = 0, todayGroups = 0;
  const recentLogs = [];

  for (let i = rows.length - 1; i >= 0; i--) {
    const r = rows[i];
    const pax = parseInt(r[4]) || 0;
    const rowDate = r[1] instanceof Date ? Utilities.formatDate(r[1], tz, 'yyyy-MM-dd') : String(r[1]);
    totalPax += pax;
    if (rowDate === todayStr) { todayPax += pax; todayGroups++; }
    if (recentLogs.length < 30) {
      recentLogs.push({ date: rowDate, time: String(r[2]), type: r[3], pax, name: r[5], phone: String(r[6]||''), interest: r[7], staff: r[8], notes: r[9] });
    }
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    stats: { todayPax, totalPax, todayGroups, totalGroups: rows.length },
    recentLogs
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const sheet = setupSheet();
  const p = JSON.parse(e.postData.contents);
  const now = new Date(), tz = Session.getScriptTimeZone();
  sheet.appendRow([
    now.toISOString(),
    Utilities.formatDate(now, tz, 'yyyy-MM-dd'),
    Utilities.formatDate(now, tz, 'HH:mm:ss'),
    p.type || 'Self Check-In',
    parseInt(p.pax) || 1,
    p.name || 'Tetamu Jemaah',
    p.phone ? "'" + p.phone : '',
    p.interest || 'Umrah',
    p.staff || '',
    p.notes || ''
  ]);
  return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
}
